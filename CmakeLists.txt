cmake_minimum_required(VERSION 3.16)
project(KVStoreCpp VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wall -Wextra -O3 -pthread")

include_directories(include)

add_library(kvstore_core
    src/store.cpp
)

add_executable(kvstore_server
    src/server.cpp
    src/main.cpp
)

target_link_libraries(kvstore_server PRIVATE kvstore_core)

# Enable testing
enable_testing()
add_executable(unit_tests tests/test_store.cpp)
target_link_libraries(unit_tests PRIVATE kvstore_core)
add_test(NAME UnitTests COMMAND unit_tests)
